    interface SweepMetric { f1: number; mrr: number; top1: number; }
    interface SweepResult { cfg: HybridWeights; train: SweepMetric; validate: SweepMetric; }

    function paretoFront(candidates: SweepResult[]): SweepResult[] {
      return candidates.filter(c => !candidates.some(other =>
        other !== c &&
        other.train.f1 >= c.train.f1 &&
        other.train.mrr >= c.train.mrr &&
        other.train.top1 >= c.train.top1 &&

    **Sensitivity helper (RESEARCH §"Pattern 3" — top1_flip_rate):**
    ```typescript
    function top1FlipRate(winnerCfg: HybridWeights, resolutions: Map<string, QueryResolution>, trainSet: CorpusEntry[]): number {
      const perturbations: HybridWeights[] = [];
      for (const axis of ["rerank", "recency", "type_match", "scope_match"] as const) {
        for (const delta of [+0.05, -0.05]) {
          const perturbed = { ...winnerCfg, [axis]: winnerCfg[axis] + delta };
          perturbations.push(perturbed);
        }
      }
      let totalFlips = 0;
      for (const perturbed of perturbations) {
        for (const entry of trainSet) {
          const res = resolutions.get(entry.id)!;
          const winnerTop1 = hybridRank(res.matches, res.blocks, {}, Date.now(), winnerCfg)[0]?.id;
          const perturbedTop1 = hybridRank(res.matches, res.blocks, {}, Date.now(), perturbed)[0]?.id;
          if (winnerTop1 !== perturbedTop1) totalFlips++;
        }
      }
      return totalFlips / (perturbations.length * trainSet.length);
    }
    ```
