       ```typescript
       import { describe, it, expect } from "vitest";
       import { env } from "cloudflare:workers";
       import { captureCallback, parseEnvelope } from "../tools-integration.test.js";   // reuse helpers
       import corpus from "./fixtures/reference-corpus.json";

       describe("AI-04 recall F1 — reference corpus (BLOCKS AI-04 closure if < 0.75)", () => {
         it("F1 ≥ 0.75 across the 20-example reference corpus", async () => {
           const ws = "ws-eval-f1";
           const rememberCb = captureCallback("remember", ws);
           const recallCb = captureCallback("recall", ws);

           // Seed: remember every example with id-traceable content
           const idMap = new Map<string, string>();
           for (const ex of corpus) {
             const res = parseEnvelope(await rememberCb({ content: ex.original_content, type: ex.expected_classified_type }, {}));
             const blockId = (res.result as any).id;
             idMap.set(ex.id, blockId);
           }
           await new Promise((r) => setTimeout(r, 5000));   // Vectorize eventual consistency

           // For each example: recall with paraphrased_query; check if intended_memory_id is in top-5
           let truePositives = 0;
           let falsePositives = 0;
           let falseNegatives = 0;

           for (const ex of corpus) {
             if (!ex.paraphrased_query) continue;   // edge cases without paraphrase skipped here
             const result = parseEnvelope(await recallCb({ query: ex.paraphrased_query, verbosity: "chunks", limit: 5 }, {}));
             const memories = (result.result as any).memories as Array<{ id: string }>;
             const expectedBlockId = idMap.get(ex.intended_memory_id);
             const isHit = memories.slice(0, 5).some((m) => m.id === expectedBlockId);
             if (isHit) {
               truePositives++;
             } else {
               falseNegatives++;
             }
             // Count false positives: memories returned that don't match expected
             falsePositives += memories.slice(0, 5).filter((m) => m.id !== expectedBlockId).length;
           }

           const precision = truePositives / (truePositives + falsePositives);
           const recall = truePositives / (truePositives + falseNegatives);
           const f1 = 2 * precision * recall / (precision + recall);

           console.log(`AI-04 F1: ${f1.toFixed(4)} (precision: ${precision.toFixed(4)}, recall: ${recall.toFixed(4)})`);
           expect(f1).toBeGreaterThanOrEqual(0.75);
         });
       });
       ```
