import type { Memory, Conflict } from "@engram/types";
import type { SqlStorage } from "@cloudflare/workers-types";
import { NotFoundError } from "./errors.js";
import type { MemoryType, InboxEntry, LexicalSearchHit } from "./types.js";

// 1. insertBlock: writes a single Memory row.
//    JSON.stringify(properties) on write per D-03.
export function insertBlock(sql: SqlStorage, block: Memory): void;

// 2. getBlock: returns Memory by id; throws NotFoundError on miss.
//    Uses .toArray() + length check (Anti-Pattern: blanket .one() try/catch masks "multiple rows" bugs).
export function getBlock(sql: SqlStorage, id: string): Memory;

// 3. lexicalSearchBlocks: WHERE content LIKE %?% OR summary LIKE %?% — naive Phase 4 backing.
//    Returns [] on no matches; the LexicalSearchHit shape extends Memory with a snippet field.
export function lexicalSearchBlocks(sql: SqlStorage, query: string, limit?: number): LexicalSearchHit[];

// 4. deleteBlock: deletes block + cascades to relations explicitly (D-04 — no ON DELETE CASCADE in DDL).
//    Returns the count of relations rows deleted (caller may log).
export function deleteBlock(sql: SqlStorage, id: string, cascade: boolean): { blocks_deleted: number; relations_deleted: number };

// 5. listMemoryTypes: returns all rows from memory_types (system + user); parses fields JSON.
export function listMemoryTypes(sql: SqlStorage): MemoryType[];

// 6. createInboxEntry: writes an InboxEntry row; JSON.stringify(proposed_properties) on write.
export function createInboxEntry(sql: SqlStorage, entry: InboxEntry): void;

// 7. listConflicts: returns Conflict[] ordered by detected_at DESC; resolved_at NULL means unresolved.
export function listConflicts(sql: SqlStorage, opts?: { resolved?: boolean; limit?: number }): Conflict[];
